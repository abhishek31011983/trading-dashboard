function initCharts(data) { // 1. Monthly Performance Chart new Chart(document.getElementById('monthlyChart'), { type: 'bar', data: { labels: data.charts.monthly.labels, datasets: [{ label: 'Monthly Closed P&L', data: data.charts.monthly.values.map(v => parseFloat(v.replace(/[₹,()]/g, '')) * (v.includes('(') ? -1 : 1)), backgroundColor: data.charts.monthly.values.map(v => v.includes('(') ? '#ef4444' : '#10b981'), borderRadius: 6 }] }, options: { responsive: true, plugins: { legend: { display: false } } } }); // 2. Segment Performance Chart new Chart(document.getElementById('segmentChart'), { type: 'doughnut', data: { labels: data.charts.segment.labels.filter(l => l), datasets: [{ data: data.charts.segment.values.filter(v => v !== '₹0.00').map(v => Math.abs(parseFloat(v.replace(/[₹,()]/g, '')))), backgroundColor: ['#6366f1', '#8b5cf6', '#ec4899', '#f43f5e', '#f59e0b', '#10b981', '#06b6d4'] }] }, options: { responsive: true } }); }
